```python
from sqlalchemy.ext.asyncio import create_async_engine, async_sessionmaker, AsyncSession
from sqlalchemy.orm import DeclarativeBase

from app.config import settings

engine = create_async_engine(settings.database_url, echo=False, pool_size=20)
async_session = async_sessionmaker(engine, class_=AsyncSession, expire_on_commit=False)


class Base(DeclarativeBase):
    pass


async def get_db() -> AsyncSession:
    async with async_session() as session:
        try:
            yield session
        finally:
            await session.close()


async def check_db():
    """启动时验证数据库连接。"""

        from sqlalchemy import text
        await conn.execute(text("SELECT 1"))
    return True
```

```python
"""Cerbos PDP 调用适配器。"""
import httpx
from app.config import settings


class CerbosAdapter:
    """封装 Cerbos PDP /api/check/resources 调用。"""

    def __init__(self):
        self._client = httpx.AsyncClient(
            base_url=settings.cerbos_pdp_url,
            timeout=10.0,
        )

    async def check_resources(
        self,
        request_id: str,
        principal: dict,
        resources: list[dict],
    ) -> dict:
        """调用 Cerbos /api/check/resources。

        resources: [{"actions": ["kb:read"], "resource": {"kind": "kb", "id": "...", "attr": {...}}}]
        """
        resp = await self._client.post(
            "/api/check/resources",
            json={
                "requestId": request_id,
                "principal": principal,
                "resources": resources,
            },
        )
        resp.raise_for_status()
        return resp.json()

    async def close(self):
        await self._client.aclose()


# 全局单例
_cerbos: CerbosAdapter | None = None


def get_cerbos() -> CerbosAdapter:
    global _cerbos
    if _cerbos is None:
        _cerbos = CerbosAdapter()
    return _cerbos
```

### 步骤 3.2 — 实现 JWT 解析与 Principal 构建

**文件**：`permission-service/services/jwt_parser.py`

```python
"""JWT 解析 — 从 credential 提取 principal 信息。"""
from jose import jwt
from app.config import settings
from app.schemas.responses import Principal


def parse_principal(credential: str) -> Principal:
    """从 JWT credential 构建 Principal 对象。
    
    包含: user_id, roles, groups, tenant_id, principals 展开。
    """
    # 本地校签 JWT
    with open(settings.jwt_public_key_path) as f:
        public_key = f.read()

    claims = jwt.decode(credential, public_key, algorithms=[settings.jwt_algorithm])

    user_id = claims.get("sub", "unknown")
    tenant_id = claims.get("tenant", claims.get("tenant_id", ""))
    roles = _norm_list(claims.get("realm_access", {}).get("roles", []))
    groups = _norm_list(claims.get("groups", []))

    # 展开 principals
    principals = [f"user:{user_id}"]
    for g in groups:
        principals.append(f"group:{g}")
    for r in roles:
        principals.append(f"role:{r}")

    return Principal(
        user_id=user_id,
        tenant_id=tenant_id,
        roles=roles,
        groups=groups,
        principals=principals,
        raw_jwt=credential,
    )


def _norm_list(val) -> list[str]:
    if isinstance(val, list):
        return val
    if isinstance(val, str):
        return [val]
    return []
```

### 步骤 3.3 — 实现 POST /v1/check

**文件**：`permission-service/api/decision.py`

**实现逻辑**（对应设计 §2.5.1）：

```
POST /v1/check
├── 1. 解析 JWT → Principal
├── 2. 查询 ACL: principal 是否直接有此权限?
├── 3. 查询 role_bindings: principal 是否通过角色绑定获得此权限?
├── 4. 查询 restrictions: 主体是否被型一封禁? 资源是否有型二限制?
├── 5. 合并上述信息 → 构造 Cerbos principal (含 granted_actions)
├── 6. 查询 resource_registry → 构造 Cerbos resource (含 retired/is_enabled/kb_id)
├── 7. POST Cerbos /api/check/resources
├── 8. 三态映射: EFFECT_ALLOW→allow, DENY→deny, 其他→indeterminate
└── 响应: {decision, decision_id, reasons}
```

**关键代码骨架**：

```python
from fastapi import APIRouter, Header, HTTPException
from app.services.jwt_parser import parse_principal
from app.services.acl_resolver import resolve_granted_actions
from app.services.cerbos_adapter import get_cerbos
from app.database import get_db

router = APIRouter(prefix="/v1", tags=["decision"])


@router.post("/check")
async def check_permission(
    request_id: str = Header(..., alias="X-Request-Id"),
    client_id: str = Header(..., alias="X-Client-Id"),
    body: dict = None,  # 实际用 Pydantic model
):
    # 1. 解析主体
    principal = parse_principal(body["credential"])

    # 2—4. 解析 ACL + 角色 + 限制
    async with async_session() as db:
        granted_actions = await resolve_granted_actions(
            db, principal,
            body["action"], body["resource"]["type"], body["resource"]["id"],
        )
        is_suspended = await check_subject_ban(db, principal)

    if is_suspended:
        return {"decision": "deny", "decision_id": request_id, "reasons": ["subject_banned"]}

    # 5. 构造 Cerbos principal
    cerbos_principal = {
        "id": f"user:{principal.user_id}",
        "roles": principal.roles + ["user"],
        "attr": {
            "tenant_id": principal.tenant_id,
            "granted_actions": granted_actions,
        },
    }

    # 6. 查 resource 属性
    async with async_session() as db:
        resource_attr = await get_resource_attr(db, body["resource"])
        resource_attr["retired"] = resource_attr.get("retired", False)
        if body.get("channel", {}).get("kb"):
            resource_attr["kb_id"] = body["channel"]["kb"]

    # 7. Cerbos 判定
    cerbos = get_cerbos()
    result = await cerbos.check_resources(
        request_id=request_id,
        principal=cerbos_principal,
        resources=[{
            "actions": [body["action"]],
            "resource": {
                "kind": body["resource"]["type"],
                "id": body["resource"]["id"],
                "attr": resource_attr,
            },
        }],
    )

    # 8. 三态映射
    item = result.get("results", [{}])[0]
    verdict = item.get("actions", {}).get(body["action"], "EFFECT_DENY")

    decision_map = {
        "EFFECT_ALLOW": "allow",
        "EFFECT_DENY": "deny",
    }
    decision = decision_map.get(verdict, "indeterminate")
    decision_id = result.get("cerbosCallId", request_id)

    return {
        "decision": decision,
        "decision_id": decision_id,
        "reasons": [],
    }
```

### 步骤 4.1 — 实现 GET /v1/prefilter

**文件**：`permission-service/api/projection.py`

**实现逻辑**（对应设计 §2.4.2）：

```
GET /v1/prefilter?credential=<JWT>
├── 1. 解析 JWT → Principal
├── 2. 查询 restrictions: 型一封禁?
│     └── 是 → {suspended: true}

├── 4. 对每个 KB:
│     └── 查询 ACL + role_bindings:
│         该 principal 是否有 kb:read 或更强权限?
│         └── 有 → 加入 kbs 列表
├── 5. 查询型二封禁的资源 → excluded_kbs
└── 响应: {kbs, excluded_kbs, tenant_wide_read, policy_version, ttl_s}
```

**性能关键**：此端点被 RAG 系统每次检索前调用，需要 p95 < 30ms。
- 对 `acl_entries(principal, resource_id)` 建立组合索引
- 对 `restrictions(principal)` 建立索引
- 缓存当前全局版本号，避免每次查 SEQUENCE

### 步骤 4.2 — 实现 POST /v1/visibility

**实现逻辑**（对应设计 §2.4.2、§2.5.2）：

```
POST /v1/visibility {tenant, doc_id, channel: {kb}}
├── 1. 查 mount_registry: 挂载存在? unlinked?
│     └── unlinked → {unmounted: true}
├── 2. 查 resource_registry: doc/kb retired?
│     └── retired → {unmounted: true}
├── 3. 查 ACL: 哪些 principal 对这个 (doc,kb) 有 doc:retrieve?
│     → allow_stamps 列表 (只含原始主体, 不展开)
├── 4. 查 restrictions: 哪些 principal 被禁止?
│     → deny_stamps 列表
├── 5. 查 role_bindings: kb_reader → allow_stamps 增加 "role:kb_reader"
└── 响应: {allow_stamps, deny_stamps, version, unmounted}
```

**性能关键**：盖戳管道高频调用，P95 < 100ms（含数据库查询 + ACL 聚合）。

### 步骤 5.2 — 实现事件发布

**文件**：`permission-service/services/event_publisher.py`

```python
"""Redis Pub/Sub 事件发布。"""
import json
import uuid
import redis.asyncio as redis
from app.config import settings


class EventPublisher:
    CHANNEL = "visibility_changed"

    def __init__(self):
        self._redis = redis.from_url(settings.redis_url)

    async def publish_visibility_changed(
        self,
        tenant_id: str,
        resource_type: str,
        resource_id: str,
        kb_id: str | None = None,
        change_detail: dict | None = None,
    ):
        """发布 VisibilityChanged 事件。"""
        from app.database import async_session
        from sqlalchemy import text

        # 获取当前全局版本号
        async with async_session() as db:
            result = await db.execute(text("SELECT nextval('global_permission_version')"))
            version = result.scalar()

            # 记录变更事件
            # INSERT INTO permission_changes ...

        event = {
            "event_id": str(uuid.uuid4()),
            "event_type": "VisibilityChanged",
            "resource": {"type": resource_type, "id": resource_id},
            "channel": {"kb": kb_id} if kb_id else None,
            "tenant": tenant_id,
            "version": version,
            "unmounted": False,
            "change_detail": change_detail or {},
            "occurred_at": None,  # Redis 侧会用服务器时间
        }

        # 移除 None 值
        event = {k: v for k, v in event.items() if v is not None}

        await self._redis.publish(self.CHANNEL, json.dumps(event))
        return version
```

```python
# 新增 Redis 订阅逻辑
import redis
import json

def subscribe_visibility_events():
    """订阅权限服务 VisibilityChanged 事件流。"""
    r = redis.from_url(settings.authz_event_stream_redis_url)
    pubsub = r.pubsub()
    pubsub.subscribe("visibility_changed")

    for message in pubsub.listen():
        if message['type'] != 'message':
            continue

        event = json.loads(message['data'])
        event_id = event['event_id']

        # 幂等去重
        if is_duplicate(event_id):
            continue

        # 处理事件
        resource_type = event['resource']['type']
        resource_id = event['resource']['id']
        kb_id = event.get('channel', {}).get('kb')
        tenant_id = event['tenant']
        unmounted = event.get('unmounted', False)

        if resource_type == 'kb':
            # KB 粒度: 展开为该 KB 下全部文档的盖戳
            on_kb_visibility_changed(resource_id, tenant_id)
        elif resource_type == 'document' and kb_id:
            # 文档粒度: 直接提交盖戳
            on_visibility_changed(resource_id, kb_id, tenant_id)

        mark_processed(event_id)
```

## 附录 B：风险与缓解

| 风险 | 影响 | 缓解措施 |
|------|------|---------|
| prefilter 性能不达标 (p95 > 30ms) | 检索延迟增加 | 组合索引优化；引入 Redis 缓存；降级为非 strict 模式 |
| visibility 大 KB 返回戳记过大 | 盖戳延迟 + Milvus 存储放大 | 戳记使用前缀通配 (user:*)；限制单个 chunk 戳记数 |
| Redis Pub/Sub 消息丢失 | 盖戳未刷新 → 权限窗口 | 使用 Redis Stream 替代 Pub/Sub；对账兜底 |
| Cerbos PDP 故障 | 全部判定失败 | 熔断器 → 拒答降级；PDP 多副本部署 |
| Keycloak 不可达 | 新登录失败 | 已签发 JWT 有效期内仍可用；管理台降级提示 |
| RAG 侧 remote 模式回退 | 切换过程中断 | AUTHZ_SERVICE_MODE 开关；local 模式保留作为 fallback |
